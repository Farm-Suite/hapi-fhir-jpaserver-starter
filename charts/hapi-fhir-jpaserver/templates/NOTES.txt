1. Obtén la URL de la aplicación ejecutando estos comandos:
{{- if .Values.global.httpRoute.enabled }}
{{- if .Values.global.httpRoute.hostnames }}
    export APP_HOSTNAME={{ .Values.global.httpRoute.hostnames | first }}
{{- else }}
    export APP_HOSTNAME="localhost"
{{- end }}
    echo "Visita https://$APP_HOSTNAME{{ .Values.http_relative_path }} para usar tu aplicación"

    NOTA: Tu HTTPRoute está configurado para HTTPS y requiere una Gateway con un listener compatible.
{{- else if contains "NodePort" .Values.service.type }}
    export NODE_PORT=$(kubectl get --namespace {{ .Release.Namespace }} -o jsonpath="{.spec.ports[0].nodePort}" services {{ include "hapi-fhir-jpaserver.fullname" . }})
    export NODE_IP=$(kubectl get nodes --namespace {{ .Release.Namespace }} -o jsonpath="{.items[0].status.addresses[0].address}")
    echo "Visita http://$NODE_IP:$NODE_PORT{{ .Values.http_relative_path }} para usar tu aplicación"
{{- else if contains "LoadBalancer" .Values.service.type }}
    export SERVICE_IP=$(kubectl get svc --namespace {{ .Release.Namespace }} {{ include "hapi-fhir-jpaserver.fullname" . }} --template "{{"{{ range (index .status.loadBalancer.ingress 0) }}{{.}}{{ end }}"}}")
    echo "Visita http://$SERVICE_IP:{{ .Values.service.port }}{{ .Values.http_relative_path }} para usar tu aplicación"
{{- else if contains "ClusterIP" .Values.service.type }}
    export POD_NAME=$(kubectl get pods --namespace {{ .Release.Namespace }} -l "app.kubernetes.io/name={{ include "hapi-fhir-jpaserver.name" . }},app.kubernetes.io/instance={{ .Release.Name }}" -o jsonpath="{.items[0].metadata.name}")
    echo "Ejecuta el siguiente comando y visita http://127.0.0.1:8080{{ .Values.http_relative_path }}"
    kubectl --namespace {{ .Release.Namespace }} port-forward $POD_NAME 8080:8080
{{- end }}
